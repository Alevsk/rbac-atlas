{{- /* Adds a "CVEs" column to the rendered Overview table (header Identity … Risk, json2hugo.py) with the per-ServiceAccount summaries of the version record ("sas"). Context: dict "html" (rendered content) "rec". Rows are matched by their "#sa-" link (sa-anchor.html); rows without data (no main workload with a valid image) get "—". Content is returned unchanged when the record has no "sas" field (data/vulns published before it existed) or the page has no Overview table. */ -}} {{- $html := .html -}} {{- if isset .rec "sas" -}} {{- $sas := .rec.sas | default dict -}} {{- $re := `(?s)\s*\s*\s*\s*\s*\s*\s*\s*\s*\s*.*?
IdentityNamespaceAutomountSecretsPermissionsWorkloadsRisk
` -}} {{- with findRE $re $html 1 -}} {{- $table := index . 0 -}} {{- $new := replaceRE `Risk(\s*)` `Risk$1CVEs` $table 1 -}} {{- $new = replaceRE `(?s)(\s*(?:)?.*?)(\s*)` `$1$3` $new -}} {{- range $name, $s := $sas -}} {{- $anchor := partialCached "vulns/sa-anchor.html" $name $name -}} {{- $cell := partialCached "vulns/sa-cell.html" $s (printf "%v" $s) -}} {{- $new = replace $new (printf `` $anchor) (printf `%s` $cell) -}} {{- end -}} {{- $new = replaceRE `` `—` $new -}} {{- $html = replace $html $table $new 1 -}} {{- end -}} {{- end -}} {{- return $html -}}
