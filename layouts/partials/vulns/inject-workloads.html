{{- /* Appends CVE summaries to the Image cells of rendered Workloads tables. Context: dict "html" (rendered content) "rec" (version record) "page". The Image column is the last cell of its row, so the needle is "RENDERED_RAW" after whitespace between and is normalized; RENDERED_RAW is the ref rendered by the page's markdown renderer (see cell-text.html). */ -}} {{- $html := .html -}} {{- with .rec.imgs -}} {{- $html = replaceRE `\s*` "" $html -}} {{- range $raw, $canonical := . -}} {{- if $raw -}} {{- $cellText := partialCached "vulns/cell-text.html" (dict "page" $.page "raw" $raw) $raw -}} {{- $needle := printf "%s" $cellText -}} {{- if in $html $needle -}} {{- $cell := partialCached "vulns/image-cell.html" ($canonical | default "") ($canonical | default "") -}} {{- $html = replace $html $needle (printf "%s%s" $cellText $cell) -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- /* Anchor the rows of riskiest-combination workloads so the callout can link to them (first match only). */ -}} {{- range .rec.combos -}} {{- $row := partial "vulns/wl-row.html" (dict "c" . "page" $.page) -}} {{- if not (in $html (printf `id="%s"` $row.id)) -}} {{- $html = replaceRE $row.pattern (printf `$1 id="%s" class="vuln-wl-row"$2` $row.id) $html 1 -}} {{- end -}} {{- end -}} {{- return $html -}}
